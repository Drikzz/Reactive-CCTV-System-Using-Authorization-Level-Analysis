# 1. See what changed (always good to check)

git status

# 2. Stage everything (or just specific files: git add filename.js)

git add .

# 3. Commit with a message (keep it short & clear)

git commit -m "fixed login bug + added dark mode"

# 4. Push to GitHub (main branch)

git push origin main

# or just: git push (if you already set upstream before with -u)
